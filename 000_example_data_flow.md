@app.route("/api/backtest/historical", methods=["POST"])
def backtest_historical_recommendations():
    try:
        data = request.get_json()
        symbol = data.get("symbol", "").upper()
        days_back = int(data.get("days_back", 30))
        
        # Query historical recommendations from database
        query = """
            SELECT id, symbol, timestamp, action, strike_price, 
                   sentiment_confidence, final_confidence, sentiment_score,
                   current_stock_price, reasoning, actual_outcome, profitable
            FROM recommendations 
            WHERE timestamp >= NOW() - INTERVAL %s days
            AND symbol = %s
            ORDER BY timestamp DESC
        """
        
        cur.execute(query, [f"{days_back} days", symbol])
        recommendations = cur.fetchall()
        
        # Process results into backtest format
        backtest_results = process_historical_recommendations(recommendations)
        
        return create_api_response(data=backtest_results)
        
    except Exception as e:
        return create_api_response(error=str(e), status_code=500)

### **Step 5: Data Processing Function**
```python
# src/web/app.py line 1339
def process_historical_recommendations(recommendations):
    """Process historical recommendations into backtest results with trade simulation."""
    
    # Initialize backtest parameters
    initial_capital = 10000  # $10,000 starting capital
    current_capital = initial_capital
    position_size = 0.02  # 2% of capital per trade
    trades = []
    cumulative_capital = [initial_capital]  # ← NEW: Track cumulative capital
    
    # Process each recommendation as a trade
    for i, rec in enumerate(recommendations):
        symbol = rec[1]  # symbol
        action = rec[4]  # action (CALL, PUT, BUY, SELL, etc.)
        confidence = float(rec[10]) if rec[10] is not None else 0.5  # final_confidence
        sentiment_score_raw = rec[11]  # sentiment_score
        current_price = float(rec[12]) if rec[12] is not None else 100
        
        # Skip HOLD actions - they don't represent actual trades
        if action == 'HOLD':
            continue
            
        # Simulate trade outcome based on sentiment and confidence
        # Higher sentiment + higher confidence = better chance of profit
        profit_probability = (sentiment_score + 1) / 2 * confidence
        
        # Simulate price movement and calculate profit
        if action in ['BUY', 'CALL']:
            price_change_pct = sentiment_score * confidence * 0.1
        else:  # SELL, PUT, SELL_SHORT
            price_change_pct = abs(sentiment_score) * confidence * 0.1
            
        # Calculate profit/loss
        trade_amount = current_capital * position_size * confidence
        shares = int(trade_amount / current_price) if current_price > 0 else 0
        profit = shares * current_price * price_change_pct
        
        # Create trade record
        trade = {
            "date": timestamp.isoformat() if timestamp else f"Trade_{i+1}",
            "action": action,
            "symbol": symbol,
            "entry_price": current_price,
            "strike_price": current_price,
            "option_price": current_price * 0.1,
            "position_size": shares,
            "cost": trade_amount,
            "exit_price": current_price * (1 + price_change_pct),
            "profit": profit,
            "sentiment": sentiment_score,
            "confidence": confidence
        }
        trades.append(trade)
        
        # Update capital and track cumulative capital
        current_capital += profit
        cumulative_capital.append(current_capital)  # ← NEW: Track for chart
    
    # Calculate statistics
    total_trades = len(trades)
    winning_trades = len([t for t in trades if t['profit'] > 0])
    win_rate = (winning_trades / total_trades * 100) if total_trades > 0 else 0
    total_return = ((current_capital - initial_capital) / initial_capital * 100)
    
    return {
        "initial_capital": initial_capital,
        "final_capital": current_capital,
        "total_return": round(total_return, 2),
        "total_trades": total_trades,
        "winning_trades": winning_trades,
        "losing_trades": total_trades - winning_trades,
        "win_rate": round(win_rate, 2),
        "trades": trades[-20:],  # Last 20 trades
        "cumulative_capital": cumulative_capital,  # ← NEW: For chart display
        "total_recommendations": len(recommendations)
    }
```
**What happens:** Raw database records are converted into simulated trade results with cumulative capital tracking

CREATE TABLE recommendations (
    id SERIAL PRIMARY KEY,
    symbol VARCHAR(10) NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    recommendation_type VARCHAR(50),
    action VARCHAR(10), -- CALL, PUT, HOLD, BUY, SELL, SELL_SHORT

    final_confidence DECIMAL(5,4),

    actual_outcome DECIMAL(5,4),
    profitable BOOLEAN
);
